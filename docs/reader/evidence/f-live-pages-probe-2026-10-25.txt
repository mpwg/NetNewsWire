let sources = [
    ("appleArticle", "https://www.apple.com/newsroom/2025/09/apple-unveils-iphone-17-pro-and-iphone-17-pro-max/", "iPhone 17 Pro"),
    ("encyclopediaArticle", "https://en.wikipedia.org/wiki/Swift_(programming_language)", "Swift"),
    ("germanHelpArticle", "https://flamedeerfaq.preisjaeger.at/help/wie-spiele-ich", "Flamedeer"),
    ("dealIndex", "https://www.preisjaeger.at/", "Deal")
]
let configuration = URLSessionConfiguration.ephemeral
configuration.timeoutIntervalForRequest = 15
configuration.timeoutIntervalForResource = 25
configuration.httpShouldSetCookies = false
configuration.urlCache = nil
let transport = URLSession(configuration: configuration)
defer { transport.invalidateAndCancel() }
let content = ArticleContentService(session: transport)
var rows: [[String: Any]] = []
for (kind, raw, marker) in sources {
    let started = Date()
    var row: [String: Any] = ["kind": kind, "url": raw]
    do {
        let result = try await content.extract(url: URL(string: raw)!)
        row["outcome"] = "extracted"
        row["textCharacters"] = result.text.count
        row["htmlBytes"] = result.html.utf8.count
        row["expectedMarkerPresent"] = result.text.localizedCaseInsensitiveContains(marker)
        row["imageElements"] = result.html.components(separatedBy: "<img").count - 1
        row["links"] = result.html.components(separatedBy: "href=").count - 1
        row["scriptElements"] = result.html.components(separatedBy: "<script").count - 1
    } catch {
        row["outcome"] = "error"
        row["error"] = String(describing: error)
    }
    row["durationSeconds"] = Date().timeIntervalSince(started)
    rows.append(row)
}
let report: [String: Any] = ["sourceCommit": "9d5fee9b998505098dce03c168459834f890b7a8", "runtime": "Native ReaderContent snippet through Xcode MCP on macOS 27.0.1", "storeWrites": false, "accountCredentialsUsed": false, "results": rows]
