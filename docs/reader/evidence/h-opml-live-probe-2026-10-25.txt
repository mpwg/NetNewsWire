func requireH(_ condition:Bool) throws { guard condition else { throw ReaderError.invalidResponse } }
func safeHError(_ error:any Error) -> String {
    if let error = error as? ReaderError {
        switch error { case .server(let status): return "server-" + String(status)
        case .invalidResponse: return "invalidResponse"
        case .invalidURL: return "invalidURL"
        default: return "ReaderError" }
    }
    if let error = error as? URLError { return "URLError-" + String(error.code.rawValue) }
    return String(reflecting:type(of:error))
}
